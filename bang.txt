    <script>
        const API_KEY_GROQ = "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";

        async function kirim() {
            const input = document.getElementById('user-input');
            const container = document.getElementById('chat-container');
            const pesan = input.value.trim();

            if (!pesan) return;

            // Tampilkan pesan user
            container.innerHTML += `<div class="msg user">${pesan}</div>`;
            input.value = "";
            container.scrollTop = container.scrollHeight;

            // Efek loading
            const loadingId = "load-" + Date.now();
            container.innerHTML += `<div class="msg bot" id="${loadingId}">Bentar, mikir...</div>`;
            container.scrollTop = container.scrollHeight;

            try {
                const response = await fetch("https://api.groq.com/openai/v1/chat/completions", {
                    method: "POST",
                    headers: {
                        "Authorization": `Bearer ${API_KEY_GROQ}`,
                        "Content-Type": "application/json"
                    },
                    body: JSON.stringify({
                        model: "llama-3.3-70b-versatile",
                        messages: [
                            {role: "system", content: "Lo adalah Bang.AI, asisten yang asik, gaul, pake bahasa tongkrongan Jakarta, tapi pinter banget."},
                            {role: "user", content: pesan}
                        ]
                    })
                });

                const data = await response.json();
                const jawaban = data.choices[0].message.content;

                // Hapus loading, ganti jawaban
                document.getElementById(loadingId).innerText = jawaban;

            } catch (e) {
                document.getElementById(loadingId).innerText = "Waduh, koneksi bermasalah atau API Key salah nih, Boss.";
            }
            container.scrollTop = container.scrollHeight;
        }

        function handleEnter(e) {
            if (e.key === 'Enter') kirim();
        }
    </script>
